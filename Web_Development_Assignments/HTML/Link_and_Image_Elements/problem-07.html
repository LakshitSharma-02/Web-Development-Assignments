    # Simple Student Management Program in Python

class Student:
    def __init__(self, name, roll_no, marks):
        self.name = name
        self.roll_no = roll_no
        self.marks = marks

    def display_details(self):
        print(f"Name: {self.name} | Roll No: {self.roll_no} | Marks: {self.marks}")

    def check_grade(self):
        if self.marks >= 90:
            return "Grade A"
        elif self.marks >= 75:
            return "Grade B"
        elif self.marks >= 50:
            return "Grade C"
        else:
            return "Needs Improvement"

# List of students
students_list = [
    Student("Aman", 101, 92),
    Student("Riya", 102, 78),
    Student("Karan", 103, 45)
]

# Display details using loop
print("--- Student Records ---")
for student in students_list:
    student.display_details()
    print("Result:", student.check_grade())
    print("-" * 25)
